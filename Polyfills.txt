
Polyfills *************
A polyfill is a piece of code (usually JavaScript) that implements modern functionality on older browsers that do not natively support it. 
It "fills in" the gap in browser support, allowing developers to use newer features without breaking the website on older or less-capable browsers.

Map ****
// Array.map((item,index,arr)=>{})

Array.Prototype.MyMap= function(cb){
	let temp=[]
	for(let i=0;i<this.length;i++){
	temp.push(cb(this[i],i,this))
	}
	return temp
}

Filter ****
// Array.filter((item,index,arr)=>{})

Array.Prototype.MyFilter= function(cb){
	let temp=[]
	for(let i=0;i<this.length;i++){
	if(cb(this[i],i,this)){
		temp.push(this[i])
		}
	}
	return temp
}

Reduce ***
// Array.reduce((acc,curr,index,arr)=>{},initialValue)

Array.Prototype.MyReducer= function(cb,initialValue){
	let accumulator=initialValue
	for(let i=0;i<this.length;i++){
		accumulator= accumulator? cb(accumulator,this[i],i,this): this[i]
	}
	return accumulator	

}


Call ****
// fn.call(context,arg1,arg2)

Function.Prototype.MyCall=function(context,...arg){
	context.fn=this
	context.fn(...arg)

}

Apply ****
// fn.apply(context,[arg1,arg2])

Function.Prototype.MyApply=function(context,arg){
	context.fn=this
	context.fn(arg)
}

Bind ********
//const bindedValue = fn.bind(context,arg1,arg2)

Function.Prototype.MyBind=function(context,...arg){
	context.fn=this
	return function (...newArgs){
		return context.fn(...arg,...newArgs)
	}
}


Promise.all ***********
Promise.allPolyfill = function (promises) {
  return new Promise((resolve, reject) => {
    const result = [];
    let count = promises.length;
    promises.forEach((promiseItem, index) => {
      Promise.resolve(promiseItem).then((res) => {
        result[index] = res;
        count--;
        if (count === 0) {
          resolve(result);
        }
      }, reject);
    });
  });
};

const p1 = Promise.resolve(3);
const p2 = 42;
const p3 = new Promise((resolve, reject) => {
  setTimeout(resolve, 1000, "foo");
});

Promise.allPolyfill([p1, p2, p3])
  .then((values) => {
    console.log(values); // [3, 42, "foo"]
  })
  .catch((error) => {
    console.error(error);
  });

function myMemoize(fn) {
  const memo = {};
  return function (...args) {
    let argsCache = JSON.stringify(args);
    if (!memo[argsCache]) {
      memo[argsCache] = fn(...args);
    }
    return memo[argsCache];
  };
}

const clumsyproduct = (num1, num2) => {
  for (let index = 0; index < 100000000; index++) {}
  return num1 * num2;
};

const newFunction = myMemoize(clumsyproduct);

console.time("First Call");
console.log("First", newFunction(121, 121));
console.timeEnd("First Call");
console.time("Second Call");
console.log("Second", newFunction(121, 121));
console.timeEnd("Second Call");

Debounce ****

const debouncedCount= debounce(()=>{ //logic},delay)

const myDebounce = (cb, delay) => {
  let timer;
  return function (...args) {
    if (timer) {
      clearTimeout(timer);
    }
    timer = setTimeout(() => {
      cb(...args);
    }, delay);
  };
};

const myThrottle = (cb, delay) => {
  let inThrottle=true;
  return function (...args) {
     if(inThrottle){
		inThrottle=false
		cb(...args)

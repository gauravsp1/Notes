  let newString = "";
  let counter = 1;
  for (let index = 0; index < inputString.length; index++) {
    if (inputString[index] === inputString[index + 1]) {
      counter++;
    } else {
      newString = newString + inputString[index] + counter;

Second Max

arr = [1, 8, 101, 7, 56, 90]

let max = -Infinity;
let secondMax = -Infinity;

arr.forEach((item) => {
  if (max < item) {
    secondMax = max;
    max = item;
  } else if (secondMax < item && max !== item) {
    secondMax = item;
  }
});
console.log("max", max, secondMax);

function CapitalFistLetter(value) {
  const result = value.toLowerCase().split(" ").map((item) => {
      return item.charAt(0).toUpperCase() + item.slice(1);
    }).join(" ");
	
  console.log(result);
}

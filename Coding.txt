In the second example, using let, the variable declared in the loop does not redeclare the variable outside the loop.
When let is used to declare the i variable in a loop, the i variable will only be visible within the loop.

Sort array- high to low,low to high *******

array.sort(sortingFunction);

Here, array is the array to be sorted, and sortingFunction is an optional function that determines the order in which the elements of the array are sorted.
If the sortingFunction is not provided, the array will be sorted in alphabetical order by default.
The sortingFunction takes two arguments and returns a value based on the comparison of the two values. 
If the return value is negative, the first value is sorted before the second value. 
If the return value is positive, the second value is sorted before the first value. 
If the return value is zero, no sorting is performed for the two values.

function fibonacci(n) {
  // Base cases: if n is 0 or 1, return n
  if (n === 0 || n === 1) {
    return n;
  }
  // Recursive case: sum of the previous two Fibonacci numbers
  return fibonacci(n - 1) + fibonacci(n - 2);
}

Output to the console will be “3”.

There are three closures in the example, each with it’s own var b declaration. When a variable is invoked closures will be checked in order 
from local to global until an instance is found. Since the inner closure has a b variable of its own, that is what will be output.

Furthermore, due to hoisting the code in inner will be interpreted as follows:

function inner () {
    var b; // b is undefined
    b++; // b is NaN
    b = 3; // b is 3
    console.log(b); // output "3"
}

 function removeVowels(str) {
  const vowels = ["a", "e", "i", "o", "u", "A", "E", "I", "O", "U"];
  let newStr = "";
  for (let i = 0; i < str.length; i++) {
    if (!vowels.includes(str[i])) {
      newStr += str[i];
    }
  }
  return newStr;
}

function convertTo24Hour(time) {
 const [timePart, modifier] = time.split(' ');
let [hours, minutes] = timePart.split(':')
 if (modifier === 'PM' && hours !== 12) {
    hours += 12;
  }
    if (modifier === 'AM' && hours === 12) {
    hours = 0;
  }
 return `${hours.toString().padStart(2, '0')}:${minutes.toString().padStart(2, '0')}`
}

}

function flatenObj(inputObj, result = {}, parent = "") {
  for (let key in inputObj) {
    if (typeof inputObj[key] === "object") {
      flatenObj(inputObj[key], result, parent + key + ".");
    } else {
      result[parent + key] = inputObj[key];
    }
  }
  return result;
}

const nestedToString = (obj) => {
  const result = [];

  const traverse = (value) => {
    if (typeof value === "object" && value !== null) {
      Object.values(value).forEach(traverse);
    } else {
      result.push(value);
    }
  };

  traverse(obj);
  return result.join(" ");
};

function setPath(obj, path, value) {
  const keys = path.split("."); // ["a", "b", "c"]
  let current = obj;

  for (let i = 0; i < keys.length; i++) {
    const key = keys[i];

    // If it's not the last key, ensure the key exists and is an object
    if (i < keys.length - 1) {
      if (typeof current[key] !== "object" || current[key] === null) {
        current[key] = {}; // create nested object if missing
      }
      current = current[key];
    } 
    // Last key — assign value
    else {
      current[key] = value;
    }
  }

  return obj; // optional, for chaining
}

function highestAverageStudent(data) {
  // Step 1: Aggregate scores using reduce
  const studentMap = data.reduce((acc, { name, score }) => {
    if (!acc[name]) acc[name] = { total: 0, count: 0 };
    acc[name].total += score;
    acc[name].count += 1;
    return acc;
  }, {});

  // Step 2: Convert to averages and find the highest
  return Object.entries(studentMap).reduce((top, [name, { total, count }]) => {
    const average = total / count;
    if (!top || average > top.average) {
      return { name, average };
    }
    return top;
  }, null);
}

function averageOccurrenceIndex(arr) {
  const map = arr.reduce((acc, num, idx) => {
    if (!acc[num]) acc[num] = { totalIndex: 0, count: 0 };
    acc[num].totalIndex += idx;
    acc[num].count += 1;
    return acc;
  }, {});

  // Convert to averages
  const result = {};
  for (const [num, { totalIndex, count }] of Object.entries(map)) {
    result[num] = Math.round(totalIndex / count);

  }

  return result;
}
